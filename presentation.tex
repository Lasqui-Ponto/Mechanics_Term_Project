\begin{frame}{1. Key Concept: 밀도 종속적 물리 법칙 제안}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{block}{카멜레온 스크리닝 효과 (Chameleon Screening)}
                스칼라 필드가 일반 물질과 결합할 때, 주변의 \textbf{물질 밀도(Matter Density)}에 따라 장의 유효 질량과 힘의 전파 거리를 조절하는 메커니즘
            \end{block}
            \vspace{0.3cm}
            \begin{itemize}
                \item \textbf{고밀도 환경}: 스칼라 필드의 유효 질량 증가 $\rightarrow$ 유효 힘의 도달 범위 수축 ($\lambda_c \approx 0$)
                \item \textbf{저밀도 환경}: 스칼라 필드의 유효 질량 감소 $\rightarrow$ 유효 힘의 도달 범위 확장 ($\lambda_c > 0$)
            \end{itemize}
        \end{column}
        
        \begin{column}{0.4\textwidth}
            \begin{block}{연구 동기}
                대안 중력 이론의 스칼라 필드가 매개하는 '제5의 힘'이 행성계 관측 및 지상 실험실 스케일에서 검출되지 않는 원인을 규명하고자 함.
            \end{block}
            \centering
            \vspace{0.2cm}
            \textcolor{TealAccent}{\textbf{밀도 의존성 역학계의 구현 및 분석}}
        \end{column}
    \end{columns}
\end{frame}

% Slide 3: The Problem (물리적 모델 및 수식 정리)

\begin{frame}{2. The Problem: 밀도 의존적 질량 및 차폐 조건}
    \begin{block}{유효 포텐셜과 지배 방정식 (Effective Potential)}
        스칼라 필드 $\phi$와 물질 밀도 $\rho_m$의 상호작용을 반영하는 유효 포텐셜 수식:
        \[
        V_{\text{eff}}(\phi) = \frac{M^{4+n}}{\phi^n} + \rho_m e^{\beta \phi / M_{\text{Pl}}}
        \]
    \end{block}
    
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{유효 질량 유도}
                유효 포텐셜 곡률로부터 유도되는 질량 $m_{\text{eff}}$:
                \[
                m_{\text{eff}}^2 = \left. \frac{d^2 V_{\text{eff}}}{d\phi^2} \right|_{\phi_{\text{min}}} \propto \rho_m^{\frac{n+2}{n+1}}
                \]
                고밀도 영역에서 필드의 질량이 비선형적으로 무거워짐을 지지함.
            \end{block}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \begin{block}{얇은 껍질 효과 (Thin-Shell Effect)}
                \begin{itemize}
                    \item 거대 천체 중심부는 필드가 고밀도 극솟값에 갇혀 $\nabla\phi \approx 0$으로 거동함.
                    \item 천체 표면 근처의 얇은 외피 두께 $\Delta R_c$의 물질만이 외부 필드 형성에 기여함.
                    \item $\beta_{\text{eff}} \approx 3 \frac{\Delta R_c}{R_c} \beta \ll \beta \rightarrow$ 제5의 힘 억제.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% ---------------------------------------------------------------------
% Slide 4: Proposed Solution (수치 해석 프로시저 설계)
% ---------------------------------------------------------------------
\begin{frame}{3. Proposed Solution: BVP 및 IVP 결합 시뮬레이션}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{카멜레온 장 수치 해석 (BVP)}
                \begin{itemize}
                    \item \textbf{1D 밀도 환경}: 시그모이드(tanh) 형태의 공간 밀도 구배 $\rho(x)$ 설정
                    \item \textbf{구형 대칭 밀도 환경}: 동심 반경별 밀도 분포 $\rho(r)$ 설정
                    \item \textbf{솔버 적용}: 정적 아인슈타인-클라인-고든 방정식
                    \[
                    \frac{d^2 \phi}{dx^2} = -n \frac{M^{4+n}}{\phi^{n+1}} + \frac{\beta}{M_{\text{Pl}}} \rho(x) e^{\beta \phi / M_{\text{Pl}}}
                    \]
                    을 \texttt{scipy.integrate.solve\_bvp}로 풀이함.
                \end{itemize}
            \end{block}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \begin{block}{입자 궤적 동역학 연산 (IVP)}
                배경 중심 복원력 및 카멜레온 제5의 힘 $F_5$ 결합:
                \[
                \frac{d^2 x}{dt^2} = -k x - \frac{\beta}{M_{\text{Pl}}} \frac{d\phi}{dx}
                \]
                의 상태 방정식을 수립하여 \texttt{solve\_ivp} (RK45)를 기반으로 궤적을 연산함.
            \end{block}
            \centering
            \vspace{0.2cm}
            \textcolor{TealAccent}{\textbf{수치 해석을 통한 궤적 분석 및 시각화}}
        \end{column}
    \end{columns}
\end{frame}
