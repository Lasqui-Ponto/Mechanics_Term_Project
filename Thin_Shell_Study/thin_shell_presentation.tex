\begin{frame}{1. The research gap is an error bar, not another screening demo}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{0.47\textwidth}
   \begin{block}{Known shortcut}
    \[
      Q_{\rm KW}=\min\!\left(3\frac{\Delta R}{R},1\right)
    \]
    One number replaces a nonlinear spherical boundary-value problem.
   \end{block}
   \vspace{0.15cm}
   \begin{itemize}
    \item fast and physically transparent
    \item derived for a thin shell and heavy core
    \item usually quoted without an accuracy estimate
   \end{itemize}
  \end{column}
  \begin{column}{0.49\textwidth}
   \begin{block}{Question}
    \centering\large
    Is the shortcut accurate enough to decide\\
    \alert{above} versus \alert{below} a force-ratio threshold?
   \end{block}
   \vspace{0.18cm}
   \begin{enumerate}
    \item Solve the full nonlinear field.
    \item Measure its scalar charge.
    \item Map approximation error.
    \item Propagate error to a proxy decision boundary.
   \end{enumerate}
  \end{column}
 \end{columns}
 \vfill
 \centering\color{graytext}\small
 Niche: an \textbf{accuracy-to-observability bridge}, not the first thin-shell calculation.
\end{frame}

\begin{frame}{2. A uniform sphere leaves two independent control parameters}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{0.54\textwidth}
   \small
   \[
    V(\phi)=\frac{\Lambda^5}{\phi},\qquad
    \nabla^2\phi=-\frac{\Lambda^5}{\phi^2}+\frac{\rho}{M_c}
   \]
   \[
    x=\frac rR,\quad u=\frac{\phi}{\phi_\infty},\quad
    \delta=\frac{\rho_c}{\rho_\infty},\quad \mu=m_\infty R
   \]
   \begin{block}{Full nonlinear BVP ($n=1$, weak-coupling limit)}
    \[
     u''+\frac{2}{x}u'=\frac{\mu^2}{2}
     \left[\widetilde\rho(x)-u^{-2}\right]
    \]
    \[
     u'(0)=0,\qquad u(\infty)=1
    \]
   \end{block}
  \end{column}
  \begin{column}{0.42\textwidth}
   \vspace{0.15cm}
   \begin{block}{Diagnostics, not extra inputs}
    \[
     \epsilon=\frac{2(1-\delta^{-1/2})}
     {\mu^2\delta}
    \]
    \[
     m_cR=\mu\delta^{3/4}
    \]
   \end{block}
   \begin{itemize}
    \item $\epsilon\ll1$: geometrically thin shell
    \item $m_cR\gg1$: core can reach its minimum
    \item $\mu\ll1$: source small compared with exterior range
   \end{itemize}
  \end{column}
 \end{columns}
 \vfill
 \scriptsize\color{graytext}
 The full exponential normalization also contains $D=\phi_\infty/M_c$; the main scan explicitly takes the same $D\to0$ limit used by the standard formula.
\end{frame}

\begin{frame}{3. The long-range law does not fix a finite-range source location}
 \small
 \begin{columns}[T,onlytextwidth]
  \begin{column}{0.31\textwidth}
   \begin{block}{Long-range law}
    \[
      \qlr=\min(3\epsilon,1),\quad \mu\ll1
    \]
    Linear shell mass plus a heuristic cap.
   \end{block}
  \end{column}
  \begin{column}{0.31\textwidth}
   \begin{block}{Two naive continuations}
    \[
      \qsurf=\qlr
    \]
    \[
      \qctr=e^{-\mu}\qlr
    \]
    Put effective charge at the surface or center.
   \end{block}
  \end{column}
  \begin{column}{0.34\textwidth}
   \begin{block}{Active-shell integral}
    \[
     g(z)=z\cosh z-\sinh z
    \]
    \[
     \qas=\frac{3e^{-\mu}}{\mu^3}
     [g(\mu)-g(\mu q)]
    \]
    Yukawa-weights the active radial shell.
   \end{block}
  \end{column}
 \end{columns}
 \vspace{0.3cm}
 \begin{block}{Convention warning}
  \centering
  Surface-referenced $Q$ carries $e^{-\mu}$; the center-referenced uniform-sphere
  form factor $3[\mu\cosh\mu-\sinh\mu]/\mu^3$ is greater than one.
 \end{block}
 \centering\color{graytext}
 $\qas$ is constructed and benchmarked here, not claimed as a literature-priority formula.
\end{frame}

\begin{frame}{4. Measuring numerical charge from the exterior tail}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{0.50\textwidth}
   \begin{block}{Charge definition}
    \[
     u=1-s\qnum\frac{e^{-\mu(x-1)}}{x},
     \qquad s=\frac{\mu^2\delta}{6}
    \]
   \end{block}
   \begin{align*}
    Q_A(x)&=\frac{x(1-u)e^{\mu(x-1)}}{s},\\
    Q_\nabla(x)&=\frac{x^2u'e^{\mu(x-1)}}{s(1+\mu x)}.
   \end{align*}
   Agreement of two estimators is a built-in diagnostic.
  \end{column}
  \begin{column}{0.46\textwidth}
   \begin{itemize}
    \item exact sharp density boundary: two collocation domains
    \item regular center and Yukawa Robin outer boundary
    \item charge-scaled variable $x(u-1)=sv$
    \item positive log-field representation in difficult transition points
    \item adaptive fourth-order collocation (SciPy \texttt{solve\_bvp})
   \end{itemize}
   \vspace{0.2cm}
   \begin{block}{Scan}
    \centering
    $25\times29=725$ points\\
    $10^2\leq\delta\leq10^5$\\
    $10^{-3}\leq\mu\leq0.3$, $x_{\max}=160$
   \end{block}
  \end{column}
 \end{columns}
 \vfill
 \centering\bfseries All 725 parameter points converged.
\end{frame}

\begin{frame}{8. An active-shell integral models most of the numerical trend}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{0.54\textwidth}
   \centering
   \setlength{\tabcolsep}{4pt}
   \begin{tabular}{lrrr}
    \toprule
     & Median & Maximum & $<10\%$ \\
    \midrule
    Surface shell & 2.78\% & 62.6\% & 68.8\% \\
    Center point & 0.566\% & 30.1\% & 91.3\% \\
    \color{blue}Active shell & \color{blue}0.391\% & \color{blue}13.1\% & \color{blue}99.7\% \\
    \bottomrule
   \end{tabular}
   \vspace{0.35cm}
   \begin{block}{Checked filter for long-range 10\% accuracy}
    \[
      \boxed{\epsilon<0.1,\quad m_cR>10,\quad m_\infty R<0.035}
    \]
    17/17 atlas + 12/12 held-out midpoints pass.
   \end{block}
  \end{column}
  \begin{column}{0.42\textwidth}
   \textbf{Two lessons}
   \begin{enumerate}
    \item $3\epsilon$ is only first order in shell thickness.
    \item Finite range requires a radial source-location model.
   \end{enumerate}
   \vspace{0.25cm}
   \begin{alertblock}{Important scope}
    ``99.7\%'' means this $n=1$, spherical, logarithmic grid.  It is not a
    proof for all chameleon potentials or geometries.
   \end{alertblock}
  \end{column}
 \end{columns}
\end{frame}

\begin{frame}{9. A force-ratio threshold is a proxy, not an apparatus reach}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{0.52\textwidth}
   \begin{block}{Outside a spherical source}
    \[
     \alpha_5\equiv\left|\frac{a_5}{a_N}\right|
     =2\beta^2Q_sQ_t(1+\mu x_p)e^{-\mu(x_p-1)}
    \]
   \end{block}
   Main map: $\beta=1$, $Q_t=1$, $x_p=1.5$.
   \vspace{0.25cm}
   \[
     \alpha_\star=10^{-2}
   \]
   is an \textbf{illustrative dimensionless threshold}, chosen to test whether
   continuation ambiguity flips a proxy decision.
  \end{column}
  \begin{column}{0.44\textwidth}
   \begin{block}{Do not conflate three statements}
    \begin{itemize}
     \item $\alpha_5$: physical signal ratio
     \item $\mathcal R=|a_5|/a_{\rm lim}$: reach of a past null result
     \item $\mathcal D=|\Delta a_5|/\sigma_a$: future SNR proxy
    \end{itemize}
   \end{block}
   Atoms can have $Q_t\simeq1$ while bulk probes screen, but different devices
   have different noise.  Scalar charge alone does not rank instruments.
  \end{column}
 \end{columns}
 \vfill
 \centering\color{graytext}\small
 No source scale or detector noise is specified: this is not a sensitivity or exclusion curve.
\end{frame}

\begin{frame}{11. What this result is---and is not}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
   \begin{block}{Defensible contribution}
    \begin{itemize}
     \item systematic spherical transition-regime error atlas
     \item verified numerical scalar-charge extraction
     \item benchmarked active-shell finite-range ansatz
     \item continuation-to-threshold ambiguity map
     \item reproducible code, data, and tests
    \end{itemize}
   \end{block}
  \end{column}
  \begin{column}{0.48\textwidth}
   \begin{alertblock}{Not claimed}
    \begin{itemize}
     \item detection of a fifth force
     \item a new physical $(\Lambda,M_c)$ exclusion
     \item validity for chambers or nonspherical sources
     \item universal accuracy or literature priority of $Q_{\rm AS}$
     \item revival of the basic dark-energy model
    \end{itemize}
   \end{alertblock}
  \end{column}
 \end{columns}
 \vspace{0.3cm}
 \small
 Chamber walls and three-dimensional geometry can change laboratory forces;
 published numerical work finds device-specific corrections at the tens-of-percent level.
 A 2022 levitated-sensor experiment reports ruling out the basic cosmologically viable
 chameleon model in its tested parameter space.  Here the model is a nonlinear mechanics
 laboratory for approximation control.
\end{frame}

\begin{frame}{12. Conclusion: continuation, radial structure, and one filter}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{0.61\textwidth}
   \begin{enumerate}
    \item \textbf{68.8\% vs 91.3\%}: surface vs center continuation within 10\%.
    \item \textbf{62.6\%}: worst surface-continuation overprediction.
    \item \textbf{99.7\%}: active-shell integral within 10\%.
   \end{enumerate}
   \vspace{0.25cm}
   \begin{block}{Practical rule from this scan}
    A small shell is not enough.  Also require a heavy core and a source small
    compared with the exterior Compton wavelength:
    \[
      \epsilon<0.1,\quad m_cR>10,\quad m_\infty R<0.035.
    \]
   \end{block}
  \end{column}
  \begin{column}{0.35\textwidth}
   \begin{block}{Research meaning}
    The long-range law alone cannot choose a finite-range source location.
    The active radial distribution controls the result.
   \end{block}
